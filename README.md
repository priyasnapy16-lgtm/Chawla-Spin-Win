function celebrate(){
  const cv=$("fx"),x=cv.getContext("2d");cv.width=innerWidth;cv.height=innerHeight;
  let ps=[],t0=performance.now(),fired=0;
  function burst(){
    const px=cv.width*(.15+Math.random()*.7),py=cv.height*(.12+Math.random()*.35),col=`hsl(${Math.random()*360},100%,60%)`;
    for(let i=0;i<70;i++){const a=Math.random()*6.283,s=1+Math.random()*4.2;ps.push({x:px,y:py,vx:Math.cos(a)*s,vy:Math.sin(a)*s,l:1,c:col});}
  }
  (function loop(now){
    const el=now-t0;
    while(fired<7&&el>fired*550){burst();fired++;}
    x.globalCompositeOperation="destination-out";x.fillStyle="rgba(0,0,0,.2)";x.fillRect(0,0,cv.width,cv.height);
    x.globalCompositeOperation="lighter";
    ps.forEach(p=>{p.x+=p.vx;p.y+=p.vy;p.vy+=.05;p.vx*=.985;p.l-=.013;x.globalAlpha=Math.max(p.l,0);x.fillStyle=p.c;x.beginPath();x.arc(p.x,p.y,2.2,0,6.283);x.fill();});
    ps=ps.filter(p=>p.l>0);
    if(fired<7||ps.length)requestAnimationFrame(loop);else x.clearRect(0,0,cv.width,cv.height);
  })(t0);
  const h=$("celebrate");h.innerHTML="";
  const sym=["✦","★","●","◆","✧"],cols=["#ffd36a","#fff2c2","#ff5fa2","#ffffff","#ff9a1f"];

    const b=document.createElement("span");b.className="confetti";b.textContent=sym[i%5];
    b.style.left=Math.random()*100+"%";b.style.color=cols[i%5];b.style.fontSize=(10+Math.random()*14)+"px";
    b.style.animationDuration=(2.4+Math.random()*2.2)+"s";b.style.animationDelay=(Math.random()*.6)+"s";h.appendChild(b);
  }
  setTimeout(()=>h.innerHTML="",5500);
}
